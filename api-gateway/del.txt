app.use('/v1/auth', proxy(process.env.IDENTITY_SERVICE_URL, {
    parseReqBody: false,  // <-- MUST BE FALSE

    proxyReqPathResolver: (req) => {
        return `/api/auth${req.url}`;
    },

    proxyReqBodyDecorator: (bodyContent, srcReq) => {
        if (srcReq.body && Object.keys(srcReq.body).length > 0) {
            const json = JSON.stringify(srcReq.body);
            return Buffer.from(json); // MUST BE BUFFER
        }
        return bodyContent;
    },

    proxyReqOptDecorator: (proxyReqOpts, srcReq) => {
        if (srcReq.body && Object.keys(srcReq.body).length > 0) {
            const json = JSON.stringify(srcReq.body);
            proxyReqOpts.headers['Content-Type'] = 'application/json';
            proxyReqOpts.headers['Content-Length'] = Buffer.byteLength(json);
        }
        return proxyReqOpts;
    },

    proxyErrorHandler: (err, res, next) => {
        res.status(500).json({ message: 'Proxy Error', error: err.message });
    }
}));
